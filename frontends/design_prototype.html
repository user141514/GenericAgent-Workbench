def quicksort(arr):
  """in-place quicksort"""
  def _partition(lo, hi):
    pivot = arr[hi]
    i = lo
    for j in range(lo, hi):
      if arr[j] <= pivot:
        arr[i], arr[j] = arr[j], arr[i]
        i += 1
    arr[i], arr[hi] = arr[hi], arr[i]
    return i
